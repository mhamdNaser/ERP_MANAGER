## [4] خدمة النطاق

`BackEnd/app/Modules/Assets/Services/AssetScopeService.php`

```php
<?php

namespace App\Modules\Assets\Services;

use App\Models\Asset;
use App\Models\User;
use Illuminate\Database\Eloquent\Builder;

class AssetScopeService
{
    public function visibleTo(User $actor): Builder
    {
        return match ($actor->primaryRole()) {
            'general_manager', 'database_manager' => Asset::query(),
            'branch_manager'  => Asset::whereHas('department', fn (Builder $q) => $q->where('branch_id', $actor->branch_id)),
            'department_head' => Asset::where('department_id', $actor->department_id),
            default           => Asset::where('holder_id', $actor->id),
        };
    }

    public function canManage(User $actor, Asset $asset): bool
    {
        return match ($actor->primaryRole()) {
            'general_manager', 'database_manager' => true,
            'branch_manager'  => $asset->department?->branch_id === $actor->branch_id,
            default           => false,
        };
    }
}
```

> **أرجع `Builder` لا `Collection`** — يسمح بإضافة فلاتر وترقيم بعدها.
> اتبع نمط `OrganizationScopeService` حرفياً.

## [6] المخرجات

`BackEnd/app/Modules/Assets/Resources/AssetResource.php`

```php
<?php

namespace App\Modules\Assets\Resources;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

class AssetResource extends JsonResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'code' => $this->code,
            'name' => $this->name,
            'status' => $this->status,
            'assigned_at' => $this->assigned_at?->toDateString(),
            'notes' => $this->notes,
            'holder'     => $this->whenLoaded('holder', fn () => ['id' => $this->holder->id, 'name' => $this->holder->name]),
            'department' => $this->whenLoaded('department'),
            'created_at' => $this->created_at?->toIso8601String(),
        ];
    }
}
```

`whenLoaded()` يمنع N+1، و`?->` يحمي من القيم الفارغة.

## [7] المتحكم

`BackEnd/app/Modules/Assets/Controllers/AssetController.php`

```php
<?php

namespace App\Modules\Assets\Controllers;

use App\Http\Controllers\Controller;
use App\Models\Asset;
use App\Modules\Assets\Requests\StoreAssetRequest;
use App\Modules\Assets\Requests\UpdateAssetRequest;
use App\Modules\Assets\Resources\AssetResource;
use App\Modules\Assets\Services\AssetScopeService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\AnonymousResourceCollection;

class AssetController extends Controller
{
    public function __construct(private AssetScopeService $scope) {}

    public function index(Request $request): AnonymousResourceCollection
    {
        return AssetResource::collection(
            $this->scope->visibleTo($request->user())->with(['holder:id,name', 'department:id,name'])->latest()->get()
        );
    }

    public function store(StoreAssetRequest $request): AssetResource
    {
        return new AssetResource(Asset::create($request->validated())->load(['holder:id,name', 'department:id,name']));
    }

    public function update(UpdateAssetRequest $request, Asset $asset): AssetResource
    {
        abort_unless($this->scope->canManage($request->user(), $asset), 403);
        $asset->update($request->validated());

        return new AssetResource($asset->load(['holder:id,name', 'department:id,name']));
    }

    public function destroy(Request $request, Asset $asset): JsonResponse
    {
        abort_unless($this->scope->canManage($request->user(), $asset), 403);
        $asset->delete();

        return response()->json(['message' => 'تم حذف العهدة.']);
    }
}
```

## [10] الصفحة

`FrontEnd/src/Page/Assets/index.jsx`

```jsx
import { useMemo, useState } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { api } from "../../lib";
import { useLanguage } from "../../Provider/LanguageContext";

export function AssetsPage({ user, notify }) {
  const { t } = useLanguage();
  const queryClient = useQueryClient();
  const [search, setSearch] = useState("");

  const { data: assets = [], isLoading } = useQuery({
    queryKey: ["assets"],
    queryFn: api.assets,
  });

  const save = useMutation({
    mutationFn: (payload) => api.createAsset(payload),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["assets"] });
      notify(t("asset_saved"), "success");
    },
    onError: (error) => notify(error.message, "error"),
  });

  const filtered = useMemo(
    () => assets.filter((a) => `${a.code} ${a.name}`.includes(search)),
    [assets, search],
  );

  const canManage = user.permissions?.includes("assets.manage");

  return (
    <div className="page">
      <header className="page-header">
        <span className="eyebrow">{t("asset_eyebrow")}</span>
        <h1 className="page-title">{t("asset_title")}</h1>
        <p className="page-subtitle">{t("asset_intro")}</p>
      </header>

      <div className="card">
        <div className="card-head">
          <input className="input" value={search} onChange={(e) => setSearch(e.target.value)}
                 placeholder={t("search")} />
        </div>
        <div className="card-body">
          <div className="table-wrap">
            <table className="table">
              <thead>
                <tr>
                  <th>{t("asset_code")}</th>
                  <th>{t("asset_name")}</th>
                  <th>{t("asset_holder")}</th>
                  <th>{t("status")}</th>
                </tr>
              </thead>
              <tbody>
                {filtered.map((asset) => (
                  <tr key={asset.id}>
                    <td>{asset.code}</td>
                    <td>{asset.name}</td>
                    <td>{asset.holder?.name ?? "—"}</td>
                    <td><span className="badge badge-neutral">{t(`asset_status_${asset.status}`)}</span></td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
          {!isLoading && filtered.length === 0 && (
            <div className="empty-state">{t("asset_empty")}</div>
          )}
        </div>
      </div>
    </div>
  );
}
```

**القواعد المطبَّقة:** أولّيات الثيم (`page`, `card`, `table`, `badge`)،
لا نص مكتوب مباشرة، لا حركة، لا خصائص فيزيائية.

> إن تجاوز الملف 400 سطر، أنشئ `Page/Assets/components/`.

## [15] الاختبار

`BackEnd/tests/Feature/AssetScopeTest.php`

```php
<?php

use App\Models\Asset;
use App\Models\Branch;
use App\Models\Department;
use App\Models\User;
use Database\Seeders\RolePermissionSeeder;
use Illuminate\Foundation\Testing\RefreshDatabase;

uses(RefreshDatabase::class);

it('limits a department head to assets in their own department', function () {
    $this->seed(RolePermissionSeeder::class);

    $branch = Branch::create(['name' => 'فرع', 'code' => 'BR1']);
    $mine   = Department::create(['name' => 'قسمي', 'code' => 'D1', 'branch_id' => $branch->id]);
    $other  = Department::create(['name' => 'قسم آخر', 'code' => 'D2', 'branch_id' => $branch->id]);

    $head = User::create([
        'name' => 'رئيس', 'email' => 'h@test.local', 'password' => 'password',
        'role' => 'department_head', 'branch_id' => $branch->id, 'department_id' => $mine->id,
        'api_token' => hash('sha256', 'head-token'),
    ]);
    $head->assignRole('department_head');

    Asset::create(['code' => 'A-1', 'name' => 'حاسوب', 'status' => 'in_service', 'department_id' => $mine->id]);
    Asset::create(['code' => 'A-2', 'name' => 'طابعة', 'status' => 'in_service', 'department_id' => $other->id]);

    $this->withToken('head-token')->getJson('/api/assets')
        ->assertOk()
        ->assertJsonCount(1)
        ->assertJsonFragment(['code' => 'A-1']);
});

it('forbids a department head from creating assets', function () {
    $this->seed(RolePermissionSeeder::class);

    $head = User::create([
        'name' => 'رئيس', 'email' => 'h2@test.local', 'password' => 'password',
        'role' => 'department_head', 'api_token' => hash('sha256', 'head2'),
    ]);
    $head->assignRole('department_head');

    $this->withToken('head2')->postJson('/api/assets', [
        'code' => 'A-9', 'name' => 'جهاز', 'status' => 'in_service',
    ])->assertForbidden();
});
```

> **احفظ الهاش وأرسل الخام** — عكسها يعطي 401 دائماً.
> **اختبر الحالة السلبية** — "لا يرى القسم الآخر" أهم من "يرى قسمه".
